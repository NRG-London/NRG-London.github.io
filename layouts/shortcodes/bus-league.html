{{- /* bus-league — the all-London bus league table. Usage: {{< bus-league >}} {{< bus-league rows=30 >}} Reads data/bus/weekly.json (hugo.Data.bus.weekly), written by the weekly sweep in E:\Road Data, and the per-route files beside it. PROGRESSIVE ENHANCEMENT, the same bargain the rail chart strikes --------------------------------------------------------------- The worst `rows` frequent routes are rendered here as real HTML, sparklines and all. A reader with JavaScript off, a crawler, or a feed reader gets a correct, house-styled, complete table rather than an empty box. `ngbus-table.js` then takes over with every route, the two tabs, sorting and search — and replaces the body at identical geometry, so nothing jumps. TWO TABLES, NOT ONE ------------------- Frequent routes and timetabled routes are different services and are judged differently, as TfL judges them. On a frequent route people turn up and wait, so the measure is excess wait. On a timetabled route people go by the timetable, so the measure is whether the bus left on time. Ranked together on excess wait, the timetabled routes came out as the best in London — the metric goes negative on exactly the routes it describes least. Night routes run outside the hours measured and are left out until they have hours of their own. Needs: /css/ngbus.css, /js/ngbus-table.js */ -}} {{- $d := hugo.Data.bus.weekly -}} {{- if not $d -}} {{- errorf "bus-league: no data/bus/weekly.json (page %s)." .Page.File.Path -}} {{- end -}} {{- $rows := int (.Get "rows" | default 20) -}} {{- $all := $d.routes -}} {{- $thr := $d.coverage_threshold -}} {{- $files := hugo.Data.bus.routes -}} {{- /* A route's kind: n(ight), t(imetabled) or f(requent). School services go by their frequency like any other route. */ -}} {{- $kindOf := dict -}} {{- range $all -}} {{- $k := "f" -}} {{- if eq .service_class "night" }}{{ $k = "n" }}{{ else if eq .frequency_type "timetabled" }}{{ $k = "t" }}{{ end -}} {{- $kindOf = merge $kindOf (dict .route $k) -}} {{- end -}} {{- /* Positions in the weekly series: the spark weeks, this week, and the week the sweep compares against. The route files are indexed positionally against `weeks`, so everything read from them goes through these. */ -}} {{- /* Spark weeks after `week_ending` are dropped. The sweep's spark runs on past it into later, partial weeks (to 30 August against a week ending 26 July), which are under the publish threshold — so every trend line ended in two blank weeks and none ended on the week the table is about. `pos` is the place in the sweep's spark array; `at` the place in `weeks`. */ -}} {{- $sparkIdx := slice }}{{ $sparkPos := slice -}} {{- range $j, $sw := $d.spark_weeks }}{{ if le $sw $d.week_ending }}{{ range $i, $wk := $d.weeks }}{{ if eq $wk $sw }}{{ $sparkIdx = $sparkIdx | append $i }}{{ $sparkPos = $sparkPos | append $j }}{{ end }}{{ end }}{{ end }}{{ end -}} {{- $cmpWeek := "" -}} {{- with where $all "delta_from" "!=" nil }}{{ $cmpWeek = (index . 0).delta_from }}{{ end -}} {{- $cmpIdx := -1 -}} {{- range $i, $wk := $d.weeks }}{{ if eq $wk $cmpWeek }}{{ $cmpIdx = $i }}{{ end }}{{ end -}} {{- /* Per route, computed once and used by both the server-rendered rows and the payload: - the sparkline with sub-threshold weeks blanked. The sweep's `spark` carries values for weeks under the publish threshold (a week at 14% collected, say), and the rule on these pages is that such a week is "no data" everywhere, the trend line included. - for timetabled routes, the on-time sparkline and change, until the sweep emits them (ask L, promised 14 Sep): the departures-weighted mean across termini, the same basis as the route-level `on_time_pct`, which it reproduces exactly. The moment a timetabled row arrives with a spark or a delta of its own, that is used instead and this goes quiet. Both the blanking and the trimming to `week_ending` are no-ops on a spark that already obeys those rules, so they can stay after the sweep's fix. */ -}} {{- $calc := newScratch -}} {{- $calc.Set "delta" (dict) -}} {{- range $all -}} {{- $r := . }}{{ $rf := index $files .route }}{{ $k := index $kindOf .route -}} {{- /* No change figure is stored rather than a nil one: a template cannot assign a bare nil, and a missing key indexes as nil anyway. */ -}} {{- $spark := slice -}} {{- $sweepSpark := false }}{{ range $r.spark }}{{ if ne . nil }}{{ $sweepSpark = true }}{{ end }}{{ end -}} {{- if and $rf $rf.termini (ne $k "n") -}} {{- $cov := (index $rf.termini 0).series -}} {{- if and (eq $k "t") (not $sweepSpark) (eq $r.delta nil) -}} {{- $idx := $sparkIdx | append $cmpIdx -}} {{- $vals := slice -}} {{- range $idx -}} {{- $at := . }}{{ $num := 0.0 }}{{ $den := 0.0 -}} {{- if ge $at 0 -}} {{- range $rf.termini -}} {{- $c := index .series $at -}} {{- if and (ne $c.on_time_pct nil) (ne $c.departures nil) (ge $c.coverage $thr) -}} {{- $num = add $num (mul $c.on_time_pct $c.departures) }}{{ $den = add $den $c.departures -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt $den 0.0 }}{{ $vals = $vals | append (div (math.Round (mul (div $num $den) 1000)) 1000) }} {{- else }}{{ $vals = $vals | append nil }}{{ end -}} {{- end -}} {{- $spark = first (len $sparkIdx) $vals -}} {{- $then := index $vals (len $sparkIdx) -}} {{- if and (ne $then nil) (ne $r.on_time_pct nil) }}{{ $calc.SetInMap "delta" $r.route (div (math.Round (mul (sub $r.on_time_pct $then) 1000)) 1000) }}{{ end -}} {{- else -}} {{- range $n, $at := $sparkIdx -}} {{- $v := index $r.spark (index $sparkPos $n) -}} {{- $spark = $spark | append (cond (ge (index $cov $at).coverage $thr) $v nil) -}} {{- end -}} {{- if ne $r.delta nil }}{{ $calc.SetInMap "delta" $r.route $r.delta }}{{ end -}} {{- end -}} {{- end -}} {{- $calc.SetInMap "spark" .route $spark -}} {{- /* The route's ends, for the line under its number. Taken from the directions the sweep now measures — "towards Victoria" gives "Victoria" — because the old terminus list also held the start of every short working and variant, and the 166 read "Banstead ↔ West Croydon ↔ Epsom Hospital ↔ Purley Cross". A route with no directions falls back to its first two termini, which are the real ends in TfL's sequence order. */ -}} {{- $ends := slice -}} {{- with $rf }}{{ range .directions -}} {{- $ends = $ends | append (strings.TrimPrefix "towards " (partial "bus-place.html" .name)) -}} {{- end }}{{ end -}} {{- if not $ends }}{{ range first 2 .termini }}{{ $ends = $ends | append (partial "bus-place.html" .name) }}{{ end }}{{ end -}} {{- $calc.SetInMap "where" .route (delimit $ends " ↔ ") -}} {{- end -}} {{- $frequent := where $all "frequency_type" "!=" "timetabled" -}} {{- $frequent = where $frequent "service_class" "!=" "night" -}} {{- $timetabled := where (where $all "frequency_type" "timetabled") "service_class" "!=" "night" -}} {{- $night := where $all "service_class" "night" -}} {{- $reporting := where $frequent "ewt_min" "!=" nil -}} {{- $ttReporting := where $timetabled "on_time_pct" "!=" nil -}} {{- $worst := first $rows (sort $reporting "ewt_min" "desc") -}} {{- /* Bar scale: the worst frequent route. */ -}} {{- $max := 1.0 -}} {{- range $reporting }}{{ if gt .ewt_min $max }}{{ $max = .ewt_min }}{{ end }}{{ end -}} {{- /* Sparkline domain, shared by every row so the shapes are comparable. A per-row domain would make a flat good route and a flat bad route draw the same picture. */ -}} {{- $slo := 0.0 }}{{ $shi := $max -}} {{- /* Curtailments arrive as an additive contract change, so every part of this table that shows them is conditional on the data actually carrying them. `where ... "!=" nil` also covers the key being absent altogether. */ -}} {{- $curted := where $all "curtailment_rate" "!=" nil -}} {{- $hasCurt := gt (len $curted) 0 -}} {{- /* The wait buses cut short add at a route's hardest-hit stop (29 Sep). Frequent routes only — on a timetabled route people go by the timetable, so the sweep leaves it null there. Same additive bargain as curtailments. */ -}} {{- $hasHit := gt (len (where $all "turned_short_worst_stop" "!=" nil)) 0 -}} {{- $window := replaceRE `(\d)-(\d)` "$1–$2" ($d.window | default "") -}} {{- /* What "on time" counted as, in words, for the timetabled column's tooltip. Taken from the data so a change of window cannot leave it quoting the old one. */ -}} {{- $otWords := "" -}} {{- with where $all "on_time_window" "!=" nil -}} {{- $nums := findRE `[0-9]+(\.[0-9]+)?` (index . 0).on_time_window -}} {{- if eq (len $nums) 2 }}{{ $otWords = printf "no more than %s minutes early or %s minutes late" (index $nums 0) (index $nums 1) }}{{ end -}} {{- end -}}
